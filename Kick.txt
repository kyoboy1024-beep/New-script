local function playSound(soundId)
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = soundId
        s.Volume = soundVolume
        s.PlayOnRemove = true
        s.Parent = SoundService
        s:Destroy()
    end)
end

local function getBlackHolePosition(obj)
    if obj:IsA("BasePart") then return obj.Position end
    local part = obj:FindFirstChildWhichIsA("BasePart", true)
    if part then return part.Position end
    return nil
end

local function getClosestPlayerToBlackHole(pos)
    local closest = nil
    local dist = math.huge
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - pos).Magnitude
                if d < dist then
                    dist = d
                    closest = plr
                end
            end
        end
    end
    return closest
end

local function applyGlowEffect(character, playerObj)
    if not character then return end
    if glowingPlayers[character] then return end
    
    local highlight = Instance.new("Highlight")
    highlight.FillColor = Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = 1
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.OutlineTransparency = 0
    highlight.Parent = character
    
    glowingPlayers[character] = {
        highlight = highlight,
        player = playerObj,
        startTime = tick()
    }
    
    task.delay(4, function()
        if glowingPlayers[character] then
            local conn
            conn = RunService.Heartbeat:Connect(function()
                local elapsed = tick() - glowingPlayers[character].startTime
                local progress = (elapsed - 4) / 1
                
                if progress >= 1 then
                    if highlight and highlight.Parent then
                        highlight:Destroy()
                    end
                    glowingPlayers[character] = nil
                    if conn then conn:Disconnect() end
                    if glowConnections[character] then
                        glowConnections[character]:Disconnect()
                        glowConnections[character] = nil
                    end
                    return
                end
                
                if highlight then
                    highlight.OutlineTransparency = progress
                end
            end)
            glowConnections[character] = conn
        end
    end)
    
    character.AncestryChanged:Connect(function(_, parent)
        if not parent then
            if highlight then highlight:Destroy() end
            glowingPlayers[character] = nil
            if glowConnections[character] then
                glowConnections[character]:Disconnect()
                glowConnections[character] = nil
            end
        end
    end)
    
    if playerObj then
        playerObj.AncestryChanged:Connect(function(_, parent)
            if not parent then
                if highlight and highlight.Parent then
                    highlight:Destroy()
                end
                glowingPlayers[character] = nil
                if glowConnections[character] then
                    glowConnections[character]:Disconnect()
                    glowConnections[character] = nil
                end
            end
        end)
    end
end

local function getPlayerFromPCLD(pcld)
    local closestPlayer
    local closestDist = 10
    
    for _, player in ipairs(Players:GetPlayers()) do
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local dist = (hrp.Position - pcld.Position).Magnitude
            if dist < closestDist then
                closestDist = dist
                closestPlayer = player
            end
        end
    end
    
    if closestPlayer then
        pcldPlayerCache[pcld] = {
            player = closestPlayer,
            time = tick(),
        }
        return closestPlayer.DisplayName .. " (@ " .. closestPlayer.Name .. ")"
    end
    
    local cached = pcldPlayerCache[pcld]
    if cached and tick() - cached.time < 2 then
        local p = cached.player
        return p.DisplayName .. " (@ " .. p.Name .. ")"
    end
    
    return "Unknown"
end

local function createSmoothPCLD(original)
    if smoothPCLDs[original] then return end
    
    original.Transparency = 1
    
    local box = Instance.new("Part")
    box.Name = "PCLD_Box"
    box.Size = original.Size
    box.CFrame = original.CFrame
    box.Anchored = true
    box.CanCollide = false
    box.CanTouch = false
    box.CanQuery = false
    box.CastShadow = false
    box.Material = Enum.Material.Neon
    box.Color = espColor
    box.Transparency = 0.45
    box.Parent = Workspace
    
    local outline = Instance.new("SelectionBox")
    outline.Adornee = box
    outline.LineThickness = 0.02
    outline.Color3 = espColor
    outline.Transparency = 0.1
    outline.Parent = box
    
    local data = {
        box = box,
        outline = outline,
        original = original,
        tween = nil,
    }
    
    smoothPCLDs[original] = data
    
    task.spawn(function()
        local lastPos = original.Position
        while box.Parent and original.Parent do
            local cf = original.CFrame
            if (original.Position - lastPos).Magnitude > 0.02 then
                lastPos = original.Position
                if data.tween then data.tween:Cancel() end
                data.tween = TweenService:Create(box, TweenInfo.new(smoothTime, Enum.EasingStyle.Linear), {CFrame = cf})
                data.tween:Play()
            end
            task.wait(0.03)
        end
    end)
end

local function removeSmoothPCLD(original)
    local d = smoothPCLDs[original]
    if not d then return end
    if d.tween then d.tween:Cancel() end
    if d.box then d.box:Destroy() end
    smoothPCLDs[original] = nil
    pcldPlayerCache[original] = nil
end

local function clearAllSmoothPCLDs()
    for _, d in pairs(smoothPCLDs) do
        if d.tween then d.tween:Cancel() end
        if d.box then d.box:Destroy() end
    end
    smoothPCLDs = {}
    pcldPlayerCache = {}
end

local function startRainbow()
    if rainbowThread then
        task.cancel(rainbowThread)
        rainbowThread = nil
    end
    
    if not rainbowPCLD or not espEnabled then return end
    
    rainbowThread = task.spawn(function()
        while rainbowPCLD and espEnabled do
            local hue = tick() % 1
            local color = Color3.fromHSV(hue, 1, 1)
            for _, d in pairs(smoothPCLDs) do
                if d.box then d.box.Color = color end
                if d.outline then d.outline.Color3 = color end
            end
            task.wait(0.05)
        end
    end)
end

local function updateColors()
    if rainbowPCLD then
        startRainbow()
        return
    end
    
    for _, d in pairs(smoothPCLDs) do
        if d.box then d.box.Color = espColor end
        if d.outline then d.outline.Color3 = espColor end
    end
end

local function enablePCLDESP()
    if espEnabled then return end
    espEnabled = true
    
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj.Name == "PlayerCharacterLocationDetector" then
            createSmoothPCLD(obj)
        end
    end
    
    cons.viewpcld = Workspace.ChildAdded:Connect(function(child)
        if child.Name == "PlayerCharacterLocationDetector" then
            task.wait(0.1)
            createSmoothPCLD(child)
        end
    end)
    
    cons.pcldRemoved = Workspace.ChildRemoved:Connect(function(child)
        if child.Name == "PlayerCharacterLocationDetector" then
            removeSmoothPCLD(child)

local function disablePCLDESP()

    rainbowPCLD = false
    
    if rainbowThread then
        task.cancel(rainbowThread)
        rainbowThread = nil
    end
    
    if cons.viewpcld then cons.viewpcld:Disconnect() end
    if cons.pcldRemoved then cons.pcldRemoved:Disconnect() end
    
    clearAllSmoothPCLDs()
end

local rs = ReplicatedStorage
local me = LocalPlayer
local Struggle = rs.CharacterEvents.Struggle
local RagdollRemote = rs.CharacterEvents.RagdollRemote

local function FWC(Parent, Name, Time)
    return Parent:FindFirstChild(Name) or Parent:WaitForChild(Name, Time or 10)
end

local function setupAntiGrab(char)
    if not char or not antiGrabEnabled then return end
    
    local hrp = FWC(char, "HumanoidRootPart")
    local hum = FWC(char, "Humanoid")
    local head = FWC(char, "Head")

    local conn1 = head.ChildAdded:Connect(function(PartOwner)
        if PartOwner.Name == "PartOwner" then
            if not antiGrabProc then
                antiGrabProc = true
                hum.Sit = false
                Struggle:FireServer(me)
                task.spawn(function()
                    while (head and head:FindFirstChild("PartOwner")) or me.IsHeld.Value do
                        Struggle:FireServer(me)
                        RagdollRemote:FireServer(hrp, 0)
                        task.wait()
                    end
                end)
                hrp.Anchored = true
                if not agWalk then
                    agWalk = true
                    while me.IsHeld.Value and task.wait() do
                        hrp.CFrame = hrp.CFrame + hum.MoveDirection * 0.43
                    end
                end
                hrp.Anchored = false
                antiGrabProc = false
                agWalk = false
            end
        end
    end)
    table.insert(antiGrabConnections, conn1)

    local ragdolled = FWC(hum, "Ragdolled")
    local conn2 = ragdolled.Changed:Connect(function()
        if hum.Ragdolled.Value then
            for _, v in pairs(char:GetChildren()) do
                if v:IsA("BasePart") and v:FindFirstChild("BallSocketConstraint") and v.Name ~= "Head" then
                    v.BallSocketConstraint.Enabled = false
                    if v:FindFirstChild("RagdollLimbPart") then
                        v.RagdollLimbPart.WeldConstraint.Enabled = false
                    end
                end
            end
        end
    end)
    table.insert(antiGrabConnections, conn2)

    local weldHRP = FWC(hrp, "WeldHRP")
    local conn3 = weldHRP.Changed:Connect(function()
        if hrp.WeldHRP.Enabled then
            while not hum.Sit do task.wait() end
            hum.Sit = false
            hum.AutoRotate = true
            hum.HipHeight = 1
            while hrp.WeldHRP.Enabled and task.wait() do
                head.CFrame = hrp.CFrame + Vector3.new(0, 1.35, 0)
            end
            hum.HipHeight = 0
        end
    end)
    table.insert(antiGrabConnections, conn3)

    for _, v in pairs(char:GetChildren()) do
        if v:IsA("BasePart") and v:FindFirstChild("BallSocketConstraint") and v.Name ~= "Head" then
            v.BallSocketConstraint.Enabled = false
            if v:FindFirstChild("RagdollLimbPart") then
                v.RagdollLimbPart.WeldConstraint.Enabled = false
            end
        end
    end
end

local function enableAntiGrab()
    if antiGrabEnabled then return end
    antiGrabEnabled = true
    antiGrabProc = false
    agWalk = false
    
    if me.Character then 
        setupAntiGrab(me.Character)

    local conn = me.CharacterAdded:Connect(function(char)

local function placePlayersInPerfectCircle(playerData, radius, yPos)
    local count = #playerData
    if count == 0 then return end
    
    local angleStep = (2 * math.pi) / count
    
    for idx, data in ipairs(playerData) do
        local angle = (idx - 1) * angleStep
        local x = radius * math.cos(angle)
        local z = radius * math.sin(angle)
        
        pcall(function()
            data.hrp.CFrame = CFrame.new(x, yPos, z)
            data.hrp.AssemblyLinearVelocity = Vector3.zero
            data.hrp.AssemblyAngularVelocity = Vector3.zero
        end)
    end
end

local function placePlayersInCross(playerData, radius, yPos)
    local count = #playerData
    if count == 0 then return end
    
    local lines = math.min(count, 4)
    local perLine = math.max(math.floor(count / lines), 1)
    local remaining = count % lines
    
    local directions = {
        0,
        math.pi / 2,
        math.pi,
        -math.pi / 2
    }
    
    local idx = 1
    for line = 1, lines do
        local countInLine = perLine + (line <= remaining and 1 or 0)
        local angle = directions[line]
        
        for pos = 1, countInLine do
            if idx <= count then
                local dist = (pos / (countInLine + 1)) * radius
                local x = dist * math.cos(angle)
                local z = dist * math.sin(angle)
                
                pcall(function()
                    playerData[idx].hrp.CFrame = CFrame.new(x, yPos, z)
                    playerData[idx].hrp.AssemblyLinearVelocity = Vector3.zero
                    playerData[idx].hrp.AssemblyAngularVelocity = Vector3.zero
                end)
                idx = idx + 1
            end
        end
    end
    
    while idx <= count do
        local angle = math.random() * 2 * math.pi
        local dist = math.random() * radius * 0.5
        local x = dist * math.cos(angle)
        local z = dist * math.sin(angle)
        
        pcall(function()
            playerData[idx].hrp.CFrame = CFrame.new(x, yPos, z)
            playerData[idx].hrp.AssemblyLinearVelocity = Vector3.zero
            playerData[idx].hrp.AssemblyAngularVelocity = Vector3.zero
        end)
        idx = idx + 1
    end
end

local function startLineLag()
    if lineLagEnabled then return end
    
    if not GrabEvents then
        OrionLib:MakeNotification({
            Name = "ERROR",
            Content = "GrabEventsが見つかりません",
            Time = 3
        })
        return
    end
    
    local createLine = GrabEvents:FindFirstChild("CreateGrabLine")
    if not createLine then
        OrionLib:MakeNotification({
            Name = "ERROR",
            Content = "CreateGrabLineが見つかりません",
            Time = 3
        })
        return
    end
    
    lineLagEnabled = true
    lineLagThread = coroutine.create(function()
        while lineLagEnabled do
            local spawnLocation = Workspace:FindFirstChild("SpawnLocation") or Workspace:FindFirstChild("Spawn") or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"))
            if spawnLocation then
                local randomX = math.random(-1.5e9, 1.5e9)
                local randomZ = math.random(-1.5e9, 1.5e9)
                local directions = {
                    CFrame.new(randomX, 0, randomZ),
                    CFrame.new(-randomX, 0, -randomZ),
                    CFrame.new(randomX, 0, -randomZ),
                    CFrame.new(-randomX, 0, randomZ),
                    CFrame.new(randomX * 1.5, 0, randomZ * 1.5),
                    CFrame.new(-randomX * 1.5, 0, -randomZ * 1.5),
                    CFrame.new(randomX * 2, 0, randomZ * 2),
                    CFrame.new(-randomX * 2, 0, -randomZ * 2),
                }
                for _, pos in pairs(directions) do 
                    pcall(function() createLine:FireServer(spawnLocation, pos) end)
                end
            end
            task.wait(0.005)
        end
    end)
    coroutine.resume(lineLagThread)
    
    OrionLib:MakeNotification({
        Name = "SERVER",
        Content = "安定強力ラグモード",
        Time = 2
    })
end

local function stopLineLag()
    lineLagEnabled = false
    if lineLagThread then 
        coroutine.close(lineLagThread) 
        lineLagThread = nil 
    end
    OrionLib:MakeNotification({
        Name = "SERVER",
        Content = "ラグ処理を停止しました",
        Time = 3
    })
end

                Content = "対象プレイヤーがいません",
                Time = 3
            })
            return
        end

        local myChar = LocalPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then 
            stopLineLag() 
            OrionLib:MakeNotification({
                Name = "ERROR",
                Content = "自分のキャラクターが見つかりません",
                Time = 3
            })
            return 
        end

        OrionLib:MakeNotification({
            Name = "ALL",
            Content = "全員のネットワークオーナーを取得中（10回）",
            Time = 2
        })

        local playerData = {}
        for _, plr in ipairs(players) do
            local char = plr.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then 
                table.insert(playerData, {player = plr, hrp = hrp}) 
            end
        end

        for _, data in ipairs(playerData) do
            teleportToPlayer(myHrp, data.hrp)
            task.wait(0.001)
            for i = 1, 10 do
                spamOwnership(data.hrp)
                task.wait(0.001)
            end
        end

        OrionLib:MakeNotification({
            Name = "ALL",
            Content = "全プレイヤーを" .. modeText .. "に配置中...",
            Time = 2
        })

        local radius = PlaceSettings.Radius
        local yPos = PlaceSettings.YPosition
        
        for _, data in ipairs(playerData) do
            pcall(function()
                data.hrp.AssemblyLinearVelocity = Vector3.zero
                data.hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
        task.wait(0.05)
        
        if PlaceSettings.PlacementMode == "Circle" then
            placePlayersInPerfectCircle(playerData, radius, yPos)
        else
            placePlayersInCross(playerData, radius, yPos)
        end
        
        task.wait(0.05)
        for _, data in ipairs(playerData) do
            pcall(function()
                data.hrp.AssemblyLinearVelocity = Vector3.zero
                data.hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
        
        for _, data in ipairs(playerData) do
            local bp = Instance.new("BodyPosition")
            bp.MaxForce = Vector3.new(2e9, 2e9, 2e9)
            bp.P = 43000000
            bp.D = 10000
            bp.Position = data.hrp.Position
            bp.Parent = data.hrp
            
            task.delay(2, function() 
                pcall(function() bp:Destroy() end) 
            end)
        end
        
        pcall(function()
            myHrp.CFrame = CFrame.new(0, yPos + 5, 0)
            myHrp.AssemblyLinearVelocity = Vector3.zero
        end)

        OrionLib:MakeNotification({
            Name = "ALL",
            Content = "ライン連打中",
            Time = 2
        })

        for i = 1, 8 do
            for _, data in ipairs(playerData) do 
                destroyLineOnPlayer(data.hrp) 
            end
            task.wait(0.2)
        end
        
        OrionLib:MakeNotification({
            Name = "ALL",
            Content = "処理完了（" .. modeText .. "配置）",
            Time = 3
        })
    end)
end
